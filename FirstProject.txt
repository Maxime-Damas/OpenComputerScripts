local robot = require("robot")

-- Helper functions for movement
local function moveRight()
  robot.turnRight()
  robot.forward()
  robot.turnLeft()
end

local function moveLeft()
  robot.turnLeft()
  robot.forward()
  robot.turnRight()
end

local function turnAround()
  robot.turnRight()
  robot.turnRight()
end


-- Move right to start the next row
local function shiftRight()
  robot.turnRight()
  robot.forward()
  robot.turnLeft()
end

-- Return to the very first row
local function shiftLeftTwice()
  robot.turnLeft()
  robot.forward()
  robot.forward()
  robot.turnRight()
end

local function prepare()
  robot.forward(2)
  moveRight()

prepare()

local function buildLayer(isMiddleLayer)
  for row = 1, 3 do
    for col = 1, 3 do
      -- The exact center of the middle layer gets the Gold Block
      if isMiddleLayer and row == 2 and col == 2 then
        robot.select(2)
      else
        robot.select(1)
      end
      
      robot.placeDown()
      
      -- Move forward for the next block, unless it's the end of the line
      if col < 3 then
        robot.forward()
      end
    end
    
    -- Reverse back to the start of the current line
    robot.back()
    robot.back()
    
    -- Shift over to start the next line
    if row < 3 then
      shiftRight()
    end
  end
  
  -- Reset position to the starting corner for the next vertical layer
  shiftLeftTwice()
end


-- Move up and forward to hover over the bottom-left corner of the build area
robot.up()
robot.forward()

-- Layer 1: Bottom (All Obsidian)
buildLayer(false)
robot.up()

-- Layer 2: Middle (Obsidian with Gold center)
buildLayer(true)
robot.up()

-- Layer 3: Top (All Obsidian)
buildLayer(false)


-- Move to the exact center of the roof
shiftRight()
robot.forward()

-- Drop the pearl directly down onto the center of the structure
robot.select(3)
robot.dropDown()

-- Return to the original starting position
robot.back()
robot.turnLeft()
robot.forward()
robot.turnRight()
robot.down()
robot.down()
robot.down()
robot.back()
